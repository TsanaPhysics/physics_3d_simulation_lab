\subsection{โค้ดแกนกลาง: สคริปต์วิเคราะห์โครงสร้าง 3 มิติและคะแนน pLDDT}

\begin{lstlisting}[language=Python, caption={AlphaFold DB API Structure Fetcher and Analyzer}]
import requests

def fetch_alphafold_structure(uniprot_id):
    url = f"https://alphafold.ebi.ac.uk/api/prediction/{uniprot_id}"
    response = requests.get(url)
    if response.status_code != 200:
        raise ValueError(f"UniProt ID {uniprot_id} not found in AlphaFold DB")
    
    data = response.json()[0]
    cif_url = data["cifUrl"]
    pdb_url = data["pdbUrl"]
    bcif_url = data["bcifUrl"]
    
    return {
        "uniprot_id": uniprot_id,
        "entry_id": data["entryId"],
        "gene": data.get("gene", "N/A"),
        "cif_url": cif_url,
        "pdb_url": pdb_url
    }
\end{lstlisting}

\subsection{โค้ดแกนกลาง: การดึงค่าความสัมพันธ์ IC50 ของเป้าหมายยา}

\begin{lstlisting}[language=Python, caption={ChEMBL Bioactivity Query Script}]
import requests

def get_target_ic50(target_chembl_id):
    endpoint = "https://www.ebi.ac.uk/chembl/api/data/activity.json"
    params = {
        "target_chembl_id": target_chembl_id,
        "standard_type": "IC50",
        "limit": 10
    }
    res = requests.get(endpoint, params=params)
    activities = res.json().get("activities", [])
    
    return [{
        "molecule_chembl_id": a["molecule_chembl_id"],
        "standard_value": a["standard_value"],
        "standard_units": a["standard_units"]
    } for a in activities if a.get("standard_value")]
\end{lstlisting}
