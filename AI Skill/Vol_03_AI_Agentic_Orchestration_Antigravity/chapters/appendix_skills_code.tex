\subsection{โค้ดแกนกลาง: วงจรตัวกรองเสียงสื่อสารอวกาศ DSP Audio Chain}

\begin{lstlisting}[language=JavaScript, caption={Web Audio API Communicator DSP Filter}]
function createJarvisAudioChain(audioCtx, sourceStream) {
  const source = audioCtx.createMediaStreamSource(sourceStream);
  
  // 1. Bandpass Filter (Radio Comm Profile 450 - 3500 Hz)
  const bandpass = audioCtx.createBiquadFilter();
  bandpass.type = "bandpass";
  bandpass.frequency.value = 1800;
  bandpass.Q.value = 1.2;
  
  // 2. High-Frequency Peaking for Cyber Presence
  const highPeak = audioCtx.createBiquadFilter();
  highPeak.type = "peaking";
  highPeak.frequency.value = 3200;
  highPeak.gain.value = 4.5;
  
  // 3. Output Dynamics Compressor
  const compressor = audioCtx.createDynamicsCompressor();
  compressor.threshold.setValueAtTime(-24, audioCtx.currentTime);
  compressor.ratio.setValueAtTime(8, audioCtx.currentTime);

  source.connect(bandpass);
  bandpass.connect(highPeak);
  highPeak.connect(compressor);
  compressor.connect(audioCtx.destination);
  
  return { bandpass, highPeak, compressor };
}
\end{lstlisting}
